프로젝트 1: 시리얼 모니터 및 NeoPixel 연동 로보카 기초 제어

1. 개요 및 주요 구현 기능

  • 시리얼 통신 기반 제어: ESP32의 Serial(115200 bps) 수신 데이터('f', 'b', 's', 'l', 'r')에 따른 모터 드라이버(TB6612FNG 계열) 방향 및 속도 제어
  • 상태 시각화 (NeoPixel): NeoPixel 2개를 활용하여 주행 상태별 직관적인 LED 피드백 제공 (전진: 초록, 후진: 빨강, 좌회전: 노랑, 우회전: 파랑, 정지: 흰색)
  • 모터 제어 최적화: STBY 핀 활성화 및 PWM 전압 제어(motorSpeed = 90)를 통한 안정적인 속도 및 제동 구현

2. 소스코드 (C++)

#include <Adafruit_NeoPixel.h>

const int PWMA = 19;
const int PWMB = 18;
const int AIN1 = 33;
const int AIN2 = 32;
const int BIN1 = 25;
const int BIN2 = 26;
const int STBY = 5;

const int LED_PIN = 27;
const int LED_COUNT = 2;
const int motorSpeed = 90;

char cmd;

Adafruit_NeoPixel strip(LED_COUNT, LED_PIN, NEO_GRB + NEO_KHZ800);

int pins[] = {PWMA, PWMB, AIN1, AIN2, BIN1, BIN2, STBY};

void setColor(int r, int g, int b) {
  for(int i = 0; i < LED_COUNT; i++) {
    strip.setPixelColor(i, strip.Color(r, g, b));
  }
  strip.show();
}

void motor(int leftSpeed, int rightSpeed, int a1, int a2, int b1, int b2) {
  analogWrite(PWMA, leftSpeed);
  analogWrite(PWMB, rightSpeed);
  digitalWrite(AIN1, a1);
  digitalWrite(AIN2, a2);
  digitalWrite(BIN1, b1);
  digitalWrite(BIN2, b2);
}

void forward() {
  motor(motorSpeed, motorSpeed, LOW, HIGH, HIGH, LOW);
  setColor(0, 255, 0);
  Serial.println("[LOG] Command 'f' Received -> Action: FORWARD | LED: GREEN");
}

void backward() {
  motor(motorSpeed, motorSpeed, HIGH, LOW, LOW, HIGH);
  setColor(255, 0, 0);
  Serial.println("[LOG] Command 'b' Received -> Action: BACKWARD | LED: RED");
}

void left() {
  motor(motorSpeed, motorSpeed, LOW, HIGH, LOW, HIGH);
  setColor(255, 255, 0);
  Serial.println("[LOG] Command 'l' Received -> Action: LEFT | LED: YELLOW");
}

void right() {
  motor(motorSpeed, motorSpeed, HIGH, LOW, HIGH, LOW);
  setColor(0, 0, 255);
  Serial.println("[LOG] Command 'r' Received -> Action: RIGHT | LED: BLUE");
}

void stopMotor() {
  motor(0, 0, LOW, LOW, LOW, LOW);
  setColor(255, 255, 255);
  Serial.println("[LOG] Command 's' Received -> Action: STOP | LED: WHITE");
}

void setup() {
  for(int i = 0; i < 7; i++) {
    pinMode(pins[i], OUTPUT);
  }
  digitalWrite(STBY, HIGH);

  strip.begin();
  strip.show();

  Serial.begin(115200);

  Serial.println("=== Motor & NeoPixel Control System ===");
  Serial.println("f : Forward  (Green)");
  Serial.println("b : Backward (Red)");
  Serial.println("s : Stop     (White)");
  Serial.println("l : Left     (Yellow)");
  Serial.println("r : Right    (Blue)");
  Serial.println("---------------------------------------");
}

void loop() {
  if(Serial.available()) {
    cmd = Serial.read();

    if(cmd == 'f') forward();
    else if(cmd == 'b') backward();
    else if(cmd == 's') stopMotor();
    else if(cmd == 'l') left();
    else if(cmd == 'r') right();
  }
}

프로젝트 2: 앱인벤터 기반 블루투스 무선 조종 앱

1. 개요 및 주요 구현 기능

  • 스마트폰 커스텀 UI 제작: 앱인벤터(App Inventor)를 활용해 직관적인 방향 제어 버튼 및 블루투스 연결 인터페이스 구축
  • 블루투스 시리얼 통신: ESP32의 BluetoothSerial을 활용하여 디바이스명 ESP32_08_CAR로 스마트폰과 1:1 무선 페어링 환경 구현
  • 실시간 제어 파싱: 수신된 단문자('f', 'b', 'l', 'r', 's')를 동기화하여 모터 제어 및 주행 상태 NeoPixel LED 피드백 연동

2. 트러블슈팅 (시행착오 및 해결 과정)

  • 문제 상황: BluetoothSerial과 WiFi 라이브러리를 동시 포함하여 컴파일 진행 시, ESP32의 기본 바이너리 용량을 초과하여 컴파일 에러(Sketch too big) 발생
  • 원인 분석: ESP32 기본 파티션 설정(Default 1.5MB)에서 두 대형 무선 스택 라이브러리가 플래시 메모리 한계를 초과함
  • 해결 방법: 아두이노 IDE 상단 메뉴의 툴(Tools) -> Partition Scheme 설정을 기존 Default에서 Huge APP (3MB No OTA/1MB SPIFFS)로 변경하여 앱 영역 메모리를 3MB로 확장 후 성공적으로 컴파일 및 업로드 완료

3. 소스코드 (C++)

#include <Adafruit_NeoPixel.h>
#include <WiFi.h>
#include <BluetoothSerial.h>

const char* ssid = "FirstClass2.4G";
const char* password = "12345678";

BluetoothSerial SerialBT;

const int PWMA = 19;
const int PWMB = 18;
const int AIN1 = 33;
const int AIN2 = 32;
const int BIN1 = 25;
const int BIN2 = 26;
const int STBY = 5;

const int LED_PIN = 27;
const int LED_COUNT = 2;
const int motorSpeed = 90;

char cmd;

Adafruit_NeoPixel strip(LED_COUNT, LED_PIN, NEO_GRB + NEO_KHZ800);

int pins[] = {PWMA, PWMB, AIN1, AIN2, BIN1, BIN2, STBY};

void setColor(int r, int g, int b) {
  for(int i = 0; i < LED_COUNT; i++) {
    strip.setPixelColor(i, strip.Color(r, g, b));
  }
  strip.show();
}

void motor(int leftSpeed, int rightSpeed, int a1, int a2, int b1, int b2) {
  analogWrite(PWMA, leftSpeed);
  analogWrite(PWMB, rightSpeed);
  digitalWrite(AIN1, a1);
  digitalWrite(AIN2, a2);
  digitalWrite(BIN1, b1);
  digitalWrite(BIN2, b2);
}

void forward() {
  motor(motorSpeed, motorSpeed, LOW, HIGH, HIGH, LOW);
  setColor(0, 255, 0);
}

void backward() {
  motor(motorSpeed, motorSpeed, HIGH, LOW, LOW, HIGH);
  setColor(255, 0, 0);
}

void left() {
  motor(motorSpeed, motorSpeed, LOW, HIGH, LOW, HIGH);
  setColor(255, 255, 0);
}

void right() {
  motor(motorSpeed, motorSpeed, HIGH, LOW, HIGH, LOW);
  setColor(0, 0, 255);
}

void stopMotor() {
  motor(0, 0, LOW, LOW, LOW, LOW);
  setColor(255, 255, 255);
}

void setup() {
  for(int i = 0; i < 7; i++) {
    pinMode(pins[i], OUTPUT);
  }
  digitalWrite(STBY, HIGH);

  strip.begin();
  strip.show();

  Serial.begin(115200);

  WiFi.begin(ssid, password);
  SerialBT.begin("ESP32_08_CAR");

  Serial.println("=== Bluetooth & WiFi Ready ===");
}

void loop() {
  if(SerialBT.available()) {
    cmd = SerialBT.read();

    if(cmd == 'f') forward();
    else if(cmd == 'b') backward();
    else if(cmd == 's') stopMotor();
    else if(cmd == 'l') left();
    else if(cmd == 'r') right();
  }
}

프로젝트 3: 웹 카메라 동작 인식 및 원격 제어 (MediaPipe)

1. 개요 및 주요 구현 기능

  • MediaPipe Hands 기반 AI 동작 인식: 웹캠 영상에서 손가락 랜드마크를 실시간 추적하고, 검지 끝 좌표(Index Finger Tip, ID 8)를 이용해 화면 속 가상 버튼과의 입체 거리를 계산하는 비접촉 터치 인터페이스 구현
  • ESP32 웹 서버 및 REST API 구축: ESP32에 고정 IP를 할당하고 웹 서버를 작동시켜 /led?state=on, /led?state=off 경로로 들어오는 HTTP 요청을 수신하는 서버 로직 개발
  • CORS 설정 및 비동기 통신: 웹 브라우저와 ESP32 웹 서버 간 Cross-Origin 이슈를 해결하기 위해 Access-Control-Allow-Origin 헤더를 설정하고, JavaScript fetch() API로 상태 변경 시에만 토글 데이터를 전송하도록 구현

2. 트러블슈팅 (시행착오 및 해결 과정)

  • 문제 상황: 손가락이 가상 버튼 영역에 머무는 동안 HTTP 요청이 프레임 단위(초당 30회 이상)로 연속 전송되어 통신 과부하 및 LED 반응 지연 발생
  • 원인 분석: 단순 영역 진입 조건만 사용 시 손이 머무르는 매 프레임마다 데이터 전송 함수가 호출됨
  • 해결 방법: 이전 프레임 상태(previousTouch)와 현재 프레임 상태(currentTouch)를 비교하는 트리거링(Edge Trigger) 로직을 도입하여, 원 밖에서 안으로 들어가는 단 한 번의 순간에만 LED 토글 명령을 1회 전송하도록 제어 안정성 확보

3. 소스코드

ESP32 아두이노 소스코드 (C++)

#include <WiFi.h>
#include <WebServer.h>
#include <Adafruit_NeoPixel.h>

const char* ssid = "FirstClass2.4G";
const char* password = "12345678";

IPAddress local_IP(10, 114, 189, 124);
IPAddress gateway(10, 114, 184, 1);
IPAddress subnet(255, 255, 252, 0);

WebServer server(80);

#define LED_PIN 27
#define LED_COUNT 2

Adafruit_NeoPixel strip(LED_COUNT, LED_PIN, NEO_GRB + NEO_KHZ800);

void ledColor(int r, int g, int b) {
  for (int i = 0; i < LED_COUNT; i++) {
    strip.setPixelColor(i, strip.Color(r, g, b));
  }
  strip.show();
}

void handleLED() {
  server.sendHeader("Access-Control-Allow-Origin", "*");
  String state = server.arg("state");

  if (state == "on") {
    ledColor(255, 0, 0);
    server.send(200, "text/plain", "LED ON");
  } else if (state == "off") {
    ledColor(0, 0, 0);
    server.send(200, "text/plain", "LED OFF");
  } else {
    server.send(400, "text/plain", "Use /led?state=on or /led?state=off");
  }
}

void handleRoot() {
  server.sendHeader("Access-Control-Allow-Origin", "*");
  server.send(200, "text/plain", "ESP32 Web Server\nLED ON : /led?state=on\nLED OFF : /led?state=off");
}

void setup() {
  Serial.begin(115200);

  strip.begin();
  strip.clear();
  strip.show();

  WiFi.mode(WIFI_STA);
  if (!WiFi.config(local_IP, gateway, subnet)) {
    Serial.println("Static IP configuration failed");
  }

  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }

  Serial.println("\nWiFi connected");
  Serial.print("ESP32 IP: ");
  Serial.println(WiFi.localIP());

  server.on("/", HTTP_GET, handleRoot);
  server.on("/led", HTTP_GET, handleLED);

  server.begin();
  Serial.println("Web Server Start");
}

void loop() {
  server.handleClient();
}

프로젝트 4: 공공데이터 Open API 기반 미세먼지 조회 대시보드

1. 개요 및 주요 구현 기능

  • 공공데이터 Open API 연동: 한국환경공단 에어코리아 대기오염정보 API를 연동하여 수원 지역 측정소의 실시간 미세먼지(PM10) 및 초미세먼지(PM2.5) 수치 데이터 수신
  • 동적 대시보드 UI 및 등급 판정: 수신된 데이터 수치 범위를 계산하여 좋음, 보통, 나쁨, 매우 나쁨 4단계 등급 텍스트와 전용 컬러 적용
  • 로컬 스토리지 연동 브라우저 상태 유지: localStorage를 활용해 공공데이터포털 인증키 및 마지막 조회 동 이름을 저장 및 자동 로드
  • 사용자 경험 개선: Enter 키를 통한 즉시 조회, decodeURIComponent 예외 처리, 데이터 결손 시 처리("자료없음" 표기) 등 예외 상황 대응 로직 구현

2. 트러블슈팅 (시행착오 및 해결 과정)

  • 문제 상황: SERVICE_KEY_IS_NOT_REGISTERED_ERROR 발생 또는 인증키의 특수문자 %가 중복 인코딩되어 데이터 요청 거부
  • 원인 분석: URLSearchParams 객체 사용 시 이미 인코딩된 키가 이중 인코딩되는 현상 발생
  • 해결 방법: 입력된 키를 decodeURIComponent()로 디코딩 후 URLSearchParams에 전달하여 API 호출 성공률 100% 확보
